<script>
    const CONFIG_KEY = 'ergotool_config';

    const DB = {
        diagnosen_key: 'ergotool_diagnosen',
        spiele_key: 'ergotool_spiele',

        getConfig() {
            const data = localStorage.getItem(CONFIG_KEY);
            return data ? JSON.parse(data) : null;
        },

        setConfig(config) {
            localStorage.setItem(CONFIG_KEY, JSON.stringify(config));
        },

        ensureConfig() {
            let config = this.getConfig();
            if (!config || !Array.isArray(config.kriterien)) {
                config = {
                    kriterien: [
                        { key: 'aufmerksamkeit', label: 'Aufmerksamkeit' },
                        { key: 'gedaechtnis', label: 'Gedächtnis' },
                        { key: 'feinmotorik', label: 'Feinmotorik' },
                        { key: 'sozial', label: 'Sozial' }
                    ]
                };
                this.setConfig(config);
            }
            return config;
        },

        getKriterien() {
            return this.ensureConfig().kriterien;
        },

        getDiagnosen() {
            const data = localStorage.getItem(this.diagnosen_key);
            const entries = data ? JSON.parse(data) : [];
            return entries.map(entry => this.fillMissingCriteria(entry));
        },

        getSpiele() {
            const data = localStorage.getItem(this.spiele_key);
            const entries = data ? JSON.parse(data) : [];
            return entries.map(entry => this.fillMissingCriteria(entry));
        },

        fillMissingCriteria(entry) {
            const result = { ...entry };
            this.getKriterien().forEach(criterion => {
                if (result[criterion.key] === undefined) {
                    result[criterion.key] = 3;
                }
            });
            return result;
        }
    };
    window.DB = DB;
